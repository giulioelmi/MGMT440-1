### Step 24: auth

- Applies only to API paths: `/^\/(stats|threads|memory|spaces)(\/|$)/`.
- Open (no header needed): `/health`, `/evals/report.json`, and all UI static files.
- Missing/blank `x-user-id` → `401 { error: 'X-User-Id header is required', status: 401, requestId }`.
- On success set `res.locals.userId`.

```ts
app.use((req, res, next) => {
  if (!/^\/(stats|threads|memory|spaces)(\/|$)/.test(req.path)) return next();
  const user = req.header(USER_HEADER)?.trim();
  if (!user) return res.status(401).json({ error: 'X-User-Id header is required', status: 401, requestId: res.locals.requestId });
  res.locals.userId = user;
  next();
});
```

```ts
const validate = (schema: ZodTypeAny): express.RequestHandler => (req, res, next) => {
  const parsed = schema.safeParse(req.body ?? {});
  if (!parsed.success) {
    return res.status(400).json({
      error: parsed.error.issues.map((i) => `${i.path.join('.') || 'body'}: ${i.message}`).join('; '),
      status: 400,
      requestId: res.locals.requestId
    });
  }
  req.body = parsed.data; // forward the parsed body: defaults (mode 'auto', depth 'quick') filled in
  next();
};
```

- Malformed JSON: in the error handler, `err.type === 'entity.parse.failed'` → `400`, not 502.

### Generic forward helper

```ts
async function forward(req, res, opts: { body?: unknown } = {}) {
  let upstream: globalThis.Response;
  try {
    upstream = await fetch(`${env.agentUrl}${req.originalUrl}`, {
      method: req.method,
      headers: {
        [USER_HEADER]: res.locals.userId ?? '',
        [REQUEST_HEADER]: res.locals.requestId,
        ...(opts.body !== undefined ? { 'content-type': 'application/json' } : {})
      },
      body: opts.body !== undefined ? JSON.stringify(opts.body) : undefined,
      signal: AbortSignal.timeout(30_000)
    });
  } catch (err) {
    return res.status(502).json({ error: `agent unreachable: ${(err as Error).message}`, status: 502, requestId: res.locals.requestId });
  }
  const text = await upstream.text();
  const status = upstream.status >= 500 ? 502 : upstream.status;
  res.status(status);
  if (text) res.type(upstream.headers.get('content-type') ?? 'application/json').send(text);
  else res.end(); // 204 from DELETE /memory/:id
}
```

**Status mapping (critical):** pass 2xx and 4xx through unchanged (201, 202, 204, 400, 404,
413, and the agent's deep-cap `429 {error, resetsAt}`). Only agent 5xx or no response → `502`.

### Upload pass-through (`POST /spaces/:spaceId/documents`)

- Do **not** parse multipart in the gateway (no multer). Stream the raw body to the agent, which uses multer.
- Keep the original `content-type` header (it contains the multipart `boundary`).
- Early `413` if `content-length > MAX_UPLOAD_BYTES`.
- Must stay fast: the bench requires the upload `202` in under 300 ms p95.

```ts
import { Readable } from 'node:stream';

async function uploadPassThrough(req, res) {
  if (Number(req.header('content-length') ?? 0) > MAX_UPLOAD_BYTES)
    return res.status(413).json({ error: `file is larger than ${MAX_UPLOAD_BYTES} bytes`, status: 413, requestId: res.locals.requestId });
  try {
    const upstream = await fetch(`${env.agentUrl}${req.originalUrl}`, {
      method: 'POST',
      headers: {
        'content-type': req.header('content-type') ?? '',
        [USER_HEADER]: res.locals.userId,
        [REQUEST_HEADER]: res.locals.requestId
      },
      body: Readable.toWeb(req) as any,
      // @ts-expect-error required by Node fetch when streaming a request body
      duplex: 'half'
    });
    const status = upstream.status >= 500 ? 502 : upstream.status;
    res.status(status).type('application/json').send(await upstream.text());
  } catch (err) {
    res.status(502).json({ error: (err as Error).message, status: 502, requestId: res.locals.requestId });
  }
}
```

### Step 27: rate limit (DECIDED, matches `DESIGN.md` trade-off #6)

Layered by cost, token buckets (not fixed windows), all in gateway memory. Four layers:

| Layer | Middleware | Applies to | Limit | Key |
|---|---|---|---|---|
| Concurrent asks | `streamLimit` | `POST /threads/:threadId/ask` | max **5** open streams | userId |
| Expensive routes | `costLimit` | ask + `POST /spaces/:spaceId/documents` | bucket **30**, refills **30/min** (`env.rateLimitPerMinute`) | userId |
| Everything else | `generalLimit` | every API route (stats, threads, memory, spaces) | bucket **120**, refills **120/min** | userId |
| IP backstop | `ipLimit` | ask + upload | bucket **120**, refills **120/min** | client IP |

Exempt (never limited): `/health`, `/evals/report.json`, static UI files, `OPTIONS` preflights.

Why these numbers (do not "simplify" to one flat 30/min limit; it fails the bench):
- `bench.mjs` `runWebWorkload` (~line 271) sends **40 asks as user `bench` at concurrency 4**
  (`sla.json` `workload.concurrency: 4`). A fixed 30/min window can 429 some of them, and the
  error-rate gate allows ≤ 1%, so a single 429 fails it. A 30-token bucket refilling 30/min
  absorbs this (30 burst + refill ≫ 40 over the run).
- While a document indexes, the bench polls `GET /spaces/:id/documents` every 1.2 s (~50/min)
  and the UI every 1.5 s (`web/src/components/SpacesPanel.tsx:45`). `generalLimit` at 120/min covers that.
- `X-User-Id` is self-asserted, so `ipLimit` is the backstop. It is generous because the bench
  and the grader send everything from one IP under several user ids (`bench`, `bench-deep-…`,
  `bench-deepcap-…`, `bench-mem-…`).
- The bench's deep-cap probe sends `cap + 2` deep asks as one user; they must reach the
  agent so the **agent** returns its `429 {error, resetsAt}`. These limits leave room for that.
- The deep-search daily cap is NOT implemented here (it stays in the agent).

`rateLimit.ts`:

```ts
import type express from 'express';
import { env } from './env.js';

type Bucket = { tokens: number; last: number };

/** Token bucket: `capacity` burst, refills `perMinute` tokens a minute. */
function tokenBucket(name: string, perMinute: number, keyOf: (req: express.Request, res: express.Response) => string) {
  const buckets = new Map<string, Bucket>();
  const ratePerMs = perMinute / 60_000;

  // Drop idle, full buckets so rotating user ids / IPs cannot grow memory forever.
  setInterval(() => {
    const now = Date.now();
    for (const [k, b] of buckets) if (now - b.last > 10 * 60_000) buckets.delete(k);
  }, 5 * 60_000).unref();

  const mw: express.RequestHandler = (req, res, next) => {
    const key = keyOf(req, res);
    const now = Date.now();
    const b = buckets.get(key) ?? { tokens: perMinute, last: now };
    b.tokens = Math.min(perMinute, b.tokens + (now - b.last) * ratePerMs);
    b.last = now;
    buckets.set(key, b);

    res.setHeader('RateLimit-Limit', String(perMinute));
    if (b.tokens >= 1) {
      b.tokens -= 1;
      res.setHeader('RateLimit-Remaining', String(Math.floor(b.tokens)));
      return next();
    }
    const waitSec = Math.ceil((1 - b.tokens) / ratePerMs / 1000);
    res.setHeader('RateLimit-Remaining', '0');
    res.setHeader('RateLimit-Reset', String(waitSec));
    res.setHeader('Retry-After', String(waitSec));
    res.status(429).json({
      error: `rate limit (${name}): ${perMinute} requests per minute`,
      status: 429,
      resetsAt: new Date(now + waitSec * 1000).toISOString(),
      requestId: res.locals.requestId
    });
  };
  return mw;
}

const userKey = (_req: express.Request, res: express.Response) => String(res.locals.userId);
// On Fly the real client IP is in Fly-Client-IP; locally fall back to req.ip.
const ipKey = (req: express.Request) => req.header('fly-client-ip') ?? req.ip ?? 'unknown';

export const generalLimit = tokenBucket('per user', env.generalRateLimitPerMinute, userKey);
export const costLimit = tokenBucket('asks and uploads', env.rateLimitPerMinute, userKey);
export const ipLimit = tokenBucket('per IP', env.ipRateLimitPerMinute, ipKey);

/** At most N open answer streams per user; the slot frees when the response closes. */
const open = new Map<string, number>();
export const streamLimit: express.RequestHandler = (_req, res, next) => {
  const user = String(res.locals.userId);
  const n = open.get(user) ?? 0;
  if (n >= env.maxConcurrentAsks) {
    res.setHeader('Retry-After', '5');
    return res.status(429).json({
      error: `too many answers in progress: at most ${env.maxConcurrentAsks} at once`,
      status: 429,
      resetsAt: new Date(Date.now() + 5000).toISOString(),
      requestId: res.locals.requestId
    });
  }
  open.set(user, n + 1);
  res.on('close', () => {
    const left = (open.get(user) ?? 1) - 1;
    if (left <= 0) open.delete(user);
    else open.set(user, left);
  });
  next();
};
```

Notes:
- `res.on('close')` fires for normal ends, errors and client disconnects, so slots never leak.
- `/health`, `/evals/report.json` and static files are exempt simply because they are not
  registered with any limiter.
- Run a bench smoke run to confirm `MAX_CONCURRENT_ASKS=5` never 429s the bench's 4 parallel asks.

### Step 28: SSE pass-through (`POST /threads/:threadId/ask`)

```ts
async function askPassThrough(req, res) {
  const abort = new AbortController();
  res.on('close', () => abort.abort()); // client left → stop the agent run

  let upstream: globalThis.Response;
  try {
    upstream = await fetch(`${env.agentUrl}${req.originalUrl}`, {
      method: 'POST',
      headers: {
        'content-type': 'application/json',
        accept: 'text/event-stream',
        [USER_HEADER]: res.locals.userId,
        [REQUEST_HEADER]: res.locals.requestId
      },
      body: JSON.stringify(req.body), // zod-parsed body
      signal: abort.signal            // NO short timeout: deep runs may take up to 240 s
    });
  } catch (err) {
    return res.status(502).json({ error: `agent unreachable: ${(err as Error).message}`, status: 502, requestId: res.locals.requestId });
  }

  // Agent refused before streaming (400 / 404 / 429 deep cap / 502): relay JSON as-is (5xx → 502).
  if (!upstream.ok || !upstream.body) {
    const status = upstream.status >= 500 ? 502 : upstream.status;
    return res.status(status).type('application/json').send(await upstream.text());
  }

  sseHeaders(res);
  const reader = upstream.body.getReader();
  try {
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      res.write(value); // raw bytes, written immediately; do not parse or re-frame
    }
  } catch {
    // Upstream broke mid-stream; headers already sent, so just end. The agent emits its own `event: error`.
  } finally {
    res.end();
  }
}
```

Never call `await upstream.text()` on a successful stream, never add compression, never set a
short timeout on this route. Buffering shows up as a failed TTFT SLA (p95 < 2500 ms).
